\begin{frame}{Low Maintenance}
	\begin{block}{Define low maintenance?}
	\begin{itemize}
		\item Durable
		\item Little to no cleaning required
		\item No calibration required after initial installation
		\item Maintenance interval measured in years
	\end{itemize}
	\end{block}
	\begin{block}{Why?}
	\begin{itemize}
		\item Unintrusive
		\item Low cost
		\item Reliability
	\end{itemize}
	\end{block}
\end{frame}
