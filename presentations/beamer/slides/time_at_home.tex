\begin{frame}{Keep Them Home Longer (1/2)}
	\begin{block}{Safety incidents force admission}
		\begin{itemize}
			\item 23\% of nursing home admissions are linked to medication non-adherence\cite{peterson2003}
			\item Presence-based auto-shutoff lets appliances be used safely unsupervised\cite{dementiacarecentral}
		\end{itemize}
	\end{block}
	\begin{block}{Slow functional and cognitive decline}
		\begin{itemize}
			\item Consistent, structured daily routines reduce agitation and confusion\cite{nccdp}\cite{ye2026}
			\item Multi-domain cognitive stimulation improves memory and executive function\cite{zou2024}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}{Keep Them Home Longer (2/2)}
	\begin{block}{Sustain caregiver/family capacity}
		\begin{itemize}

			\item In-home monitoring protects caregiver sleep efficiency\cite{scheffer2024}
		\end{itemize}
	\end{block}
	\begin{block}{Enable timely interventions}
		\begin{itemize}

		\end{itemize}
	\end{block}
\end{frame}
